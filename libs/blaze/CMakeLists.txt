file(GLOB_RECURSE blaze_sources CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c"
)

if(blaze_sources)
    add_library(blaze STATIC ${blaze_sources})
    target_include_directories(blaze PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(blaze PRIVATE MergeSource::Options PUBLIC Nexus::Nexus)

    if(UNIX)
        target_link_libraries(blaze PUBLIC m)
    endif()
else()
    add_library(blaze INTERFACE)
    target_include_directories(blaze INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(blaze INTERFACE Nexus::Nexus)

    if(UNIX)
        target_link_libraries(blaze INTERFACE m)
    endif()
endif()

add_library(Blaze::Blaze ALIAS blaze)

if(BUILD_TESTING)
    add_executable(blaze_color_tests tests/b_color_tests.c)
    target_link_libraries(blaze_color_tests PRIVATE MergeSource::Options Blaze::Blaze)
    add_test(NAME blaze_color_tests COMMAND blaze_color_tests)
    add_executable(blaze_geometry_tests tests/b_geometry_tests.c)
    target_link_libraries(blaze_geometry_tests PRIVATE MergeSource::Options Blaze::Blaze)
    add_test(NAME blaze_geometry_tests COMMAND blaze_geometry_tests)

    foreach(vector_precision IN ITEMS 32 64)
        set(vector_test_target "blaze_vectors_f${vector_precision}_tests")
        add_executable(${vector_test_target} tests/b_vectors_tests.c)
        target_link_libraries(${vector_test_target} PRIVATE MergeSource::Options Blaze::Blaze)
        if(vector_precision EQUAL 64)
            target_compile_definitions(${vector_test_target} PRIVATE NEXUS_FLOAT_DOUBLE_PRECISION=1)
        else()
            target_compile_definitions(${vector_test_target} PRIVATE NEXUS_FLOAT_DOUBLE_PRECISION=0)
        endif()
        add_test(NAME ${vector_test_target} COMMAND ${vector_test_target})

        set(numerics_test_target "blaze_numerics_f${vector_precision}_tests")
        add_executable(${numerics_test_target} tests/b_numerics_tests.c)
        target_link_libraries(${numerics_test_target} PRIVATE MergeSource::Options Blaze::Blaze)
        if(vector_precision EQUAL 64)
            target_compile_definitions(${numerics_test_target} PRIVATE NEXUS_FLOAT_DOUBLE_PRECISION=1)
        else()
            target_compile_definitions(${numerics_test_target} PRIVATE NEXUS_FLOAT_DOUBLE_PRECISION=0)
        endif()
        add_test(NAME ${numerics_test_target} COMMAND ${numerics_test_target})
    endforeach()
endif()
